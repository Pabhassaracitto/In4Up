Option C: Hybrid (Intelligent Switching)
─────────────────────────────────────────
Tự động chuyển V1/V2 dựa trên speed:

    class SmartEngine {
    public:
        SmartEngine() {
            engineV1_ = std::make_unique<Engine>();
            engineV2_ = std::make_unique<V2::EngineV2>();
            currentVersion_ = 1;
        }
        
        void initialize(int sampleRate, int channels) {
            engineV1_->initialize(sampleRate, channels, 
                Options{Quality::Standard});
            engineV2_->initialize(sampleRate, channels, 
                Options{Quality::UltraQuality});
        }
        
        void setSpeed(float speed) {
            speed_ = speed;
            
            // Switch to V2 for extreme slow
            if (speed < 0.15f && currentVersion_ == 1) {
                std::cout << "Switching to V2 for better quality\n";
                currentVersion_ = 2;
            }
            // Switch back to V1 for normal speeds
            else if (speed >= 0.3f && currentVersion_ == 2) {
                std::cout << "Switching to V1 for efficiency\n";
                currentVersion_ = 1;
            }
            
            engineV1_->setSpeed(speed);
            engineV2_->setSpeed(speed);
        }
        
        int process(const float* input, int inputSize,
                   float* output, int maxOutputSize) {
            if (currentVersion_ == 2) {
                return engineV2_->processV2(input, inputSize, 
                                           output, maxOutputSize);
            } else {
                return engineV1_->process(input, inputSize, 
                                         output, maxOutputSize);
            }
        }
        
    private:
        std::unique_ptr<Engine> engineV1_;
        std::unique_ptr<V2::EngineV2> engineV2_;
        float speed_;
        int currentVersion_;
    };

CMakeLists.txt:
───────────────

    cmake_minimum_required(VERSION 3.10)
    project(UltraTimeStretch)
    
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    
    # Optimization flags
    if(CMAKE_BUILD_TYPE MATCHES Release)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -march=native")
    endif()
    
    # SIMD detection
    if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64")
        add_definitions(-DUTS_USE_NEON=1)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
        add_definitions(-DUTS_USE_SSE=1)
        set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -msse2 -msse3")
    endif()
    
    # V1 Library
    add_library(UltraTimeStretchV1 STATIC
        Core/FFTProcessor.cpp
        Core/TransientDetector.cpp
        Core/PhaseVocoder_Enhanced.cpp
        Core/WSOLAProcessor.cpp
        Core/HybridStretcher.cpp
        Core/Engine.cpp
    )
    
    # V2 Library (optional)
    option(BUILD_V2 "Build V2 enhanced version" ON)
    if(BUILD_V2)
        add_library(UltraTimeStretchV2 STATIC
            V2/MultiResolutionPV.cpp
            V2/HarmonicPercussiveSeparator.cpp
            V2/FormantPreserver.cpp
            V2/SpectralPeakInterpolator.cpp
        )
        target_link_libraries(UltraTimeStretchV2 UltraTimeStretchV1)
    endif()
    
    # Example executable
    add_executable(ExampleV1 Examples/Example.cpp)
    target_link_libraries(ExampleV1 UltraTimeStretchV1)
    
    if(BUILD_V2)
        add_executable(ExampleV2 Examples/Example_V2.cpp)
        target_link_libraries(ExampleV2 UltraTimeStretchV2)
        
        add_executable(Benchmark Examples/Benchmark_V1_vs_V2.cpp)
        target_link_libraries(Benchmark UltraTimeStretchV2)
    endif()

iOS Swift Integration:
──────────────────────

    import Foundation
    
    class AudioProcessor {
        private var engine: OpaquePointer?
        
        init(sampleRate: Int, channels: Int) {
            // Call C++ constructor via wrapper
            engine = UTS_createEngine()
            UTS_initialize(engine, Int32(sampleRate), Int32(channels))
        }
        
        func setSpeed(_ speed: Float) {
            UTS_setSpeed(engine, speed)
        }
        
        func process(input: [Float]) -> [Float] {
            var output = [Float](repeating: 0, count: input.count * 10)
            let outSamples = UTS_process(
                engine,
                input, Int32(input.count),
                &output, Int32(output.count)
            )
            return Array(output.prefix(Int(outSamples)))
        }
        
        deinit {
            UTS_destroy(engine)
        }
    }


Android Kotlin Integration:
────────────────────────────

    class AudioProcessor(sampleRate: Int, channels: Int) {
        private var nativeEngine: Long = 0
        
        init {
            System.loadLibrary("UltraTimeStretch")
            nativeEngine = createEngine(sampleRate, channels)
        }
        
        fun setSpeed(speed: Float) {
            setSpeedNative(nativeEngine, speed)
        }
        
        fun process(input: FloatArray): FloatArray {
            val output = FloatArray(input.size * 10)
            val outSize = processNative(
                nativeEngine, input, output
            )
            return output.copyOf(outSize)
        }
        
        private external fun createEngine(rate: Int, ch: Int): Long
        private external fun setSpeedNative(engine: Long, speed: Float)
        private external fun processNative(
            engine: Long, input: FloatArray, output: FloatArray
        ): Int
        
        protected fun finalize() {
            destroyEngine(nativeEngine)
        }
    }

🎓 TROUBLESHOOTING GUIDE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Problem: Audio sounds "robotic" at 0.1x
Solution: Enable UltraQuality mode and formant preservation

Problem: CPU usage too high
Solution: Reduce FFT size or use V1 instead of V2

Problem: Crashes on mobile
Solution: Check memory limits, reduce quality settings

Problem: Artifacts/clicking sounds
Solution: Increase overlap factor or enable smoothTransitions

Problem: Build fails with SIMD errors
Solution: Disable SIMD with -DUTS_USE_NEON=0
